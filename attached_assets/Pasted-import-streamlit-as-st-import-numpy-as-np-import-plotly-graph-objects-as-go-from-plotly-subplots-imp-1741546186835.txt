import streamlit as st
import numpy as np
import plotly.graph_objects as go
from plotly.subplots import make_subplots

# Configuration
PLOT_HEIGHT = 500
PLOT_WIDTH = 700
COLOR_SCHEME = {
    'superconducting': 'blue',
    'normal': 'red',
    'mixed': 'purple'
}

st.title("Superconductivity")

st.markdown("""
Explore the fascinating world of superconductivity, where materials exhibit zero electrical 
resistance and expel magnetic fields when cooled below a critical temperature.
""")

# Sidebar controls
st.sidebar.header("Superconductivity Parameters")

# Material selection
material = st.sidebar.selectbox(
    "Select Material",
    ["Lead (Pb)", "Niobium-Titanium (NbTi)", "YBCO", "Mercury (Hg)", "Aluminum (Al)"]
)

# Material properties (values are approximate)
materials = {
    "Lead (Pb)": {"type": "Type-I", "tc": 7.2, "hc": 0.08, "penetration_depth": 39},
    "Niobium-Titanium (NbTi)": {"type": "Type-II", "tc": 9.8, "hc1": 0.01, "hc2": 14.5, "penetration_depth": 300},
    "YBCO": {"type": "Type-II", "tc": 93, "hc1": 0.01, "hc2": 120, "penetration_depth": 150},
    "Mercury (Hg)": {"type": "Type-I", "tc": 4.2, "hc": 0.04, "penetration_depth": 40},
    "Aluminum (Al)": {"type": "Type-I", "tc": 1.2, "hc": 0.01, "penetration_depth": 50}
}

try:
    # Get current material properties
    current_material = materials[material]
    material_type = current_material["type"]
    tc = current_material["tc"]

    # Temperature control
    temperature = st.sidebar.slider(
        "Temperature (K)", 
        0.0, 
        float(tc * 1.5), 
        float(tc / 2),
        step=0.1,
        help="Adjust temperature to see effect on superconductivity"
    )
    
    if temperature < 0:
        st.sidebar.warning("Temperature cannot be negative. Setting to 0.")
        temperature = 0

    # Magnetic field control
    if material_type == "Type-I":
        hc = current_material["hc"]
        h_max = hc * 2
        magnetic_field = st.sidebar.slider(
            "Magnetic Field (Tesla)", 
            0.0, 
            float(h_max), 
            0.0,
            step=0.01,
            help="Adjust magnetic field strength"
        )
    else:  # Type-II
        hc1 = current_material["hc1"]
        hc2 = current_material["hc2"]
        h_max = hc2 * 1.2
        magnetic_field = st.sidebar.slider(
            "Magnetic Field (Tesla)", 
            0.0, 
            float(h_max), 
            0.0,
            step=0.1,
            help="Adjust magnetic field strength"
        )

    # Main interface
    st.header(f"{material} ({material_type})")

    # Information about the material
    st.markdown(f"""
    **Key Properties:**
    - Critical Temperature (Tc): {tc} K
    - Type: {material_type}
    """)

    if material_type == "Type-I":
        st.markdown(f"- Critical Field (Hc): {current_material['hc']} Tesla")
    else:
        st.markdown(f"""
        - Lower Critical Field (Hc1): {current_material['hc1']} Tesla
        - Upper Critical Field (Hc2): {current_material['hc2']} Tesla
        """)

    st.markdown(f"- London Penetration Depth: {current_material['penetration_depth']} nm")

    # Create tabs for visualizations
    with st.spinner("Generating visualizations..."):
        phase_tab, meissner_tab, cooper_tab = st.tabs(["Phase Diagram", "Meissner Effect", "Cooper Pairs"])

        with phase_tab:
            st.subheader("H-T Phase Diagram")
            if material_type == "Type-I":
                temperatures = np.linspace(0, tc * 1.2, 100)
                critical_fields = current_material["hc"] * (1 - (temperatures/tc)**2)
                phase_fig = go.Figure()
                phase_fig.add_trace(go.Scatter(
                    x=temperatures, 
                    y=critical_fields,
                    mode='lines',
                    name='Critical Field',
                    line=dict(color='blue', width=3)
                ))
                phase_fig.add_trace(go.Scatter(
                    x=np.concatenate([temperatures, temperatures[::-1]]),
                    y=np.concatenate([np.zeros_like(temperatures), critical_fields[::-1]]),
                    fill='toself',
                    fillcolor='rgba(0, 0, 255, 0.2)',
                    line=dict(color='rgba(0, 0, 255, 0)'),
                    name='Superconducting'
                ))
                phase_fig.add_trace(go.Scatter(
                    x=np.concatenate([temperatures, [tc*1.2, 0]]),
                    y=np.concatenate([critical_fields, [0, current_material["hc"]]]),
                    fill='toself',
                    fillcolor='rgba(255, 0, 0, 0.2)',
                    line=dict(color='rgba(255, 0, 0, 0)'),
                    name='Normal'
                ))
                if temperature <= tc and magnetic_field <= current_material["hc"] * (1 - (temperature/tc)**2):
                    state = "Superconducting"
                    state_color = "blue"
                else:
                    state = "Normal"
                    state_color = "red"
                phase_fig.add_trace(go.Scatter(
                    x=[temperature],
                    y=[magnetic_field],
                    mode='markers',
                    marker=dict(size=15, color=state_color, symbol='star'),
                    name='Current State'
                ))
            else:
                temperatures = np.linspace(0, tc * 1.2, 100)
                hc1_values = current_material["hc1"] * (1 - (temperatures/tc)**2)
                hc2_values = current_material["hc2"] * (1 - (temperatures/tc)**2)
                phase_fig = go.Figure()
                phase_fig.add_trace(go.Scatter(
                    x=temperatures, 
                    y=hc1_values,
                    mode='lines',
                    name='Lower Critical Field (Hc1)',
                    line=dict(color='blue', width=3)
                ))
                phase_fig.add_trace(go.Scatter(
                    x=temperatures, 
                    y=hc2_values,
                    mode='lines',
                    name='Upper Critical Field (Hc2)',
                    line=dict(color='red', width=3)
                ))
                phase_fig.add_trace(go.Scatter(
                    x=np.concatenate([temperatures, temperatures[::-1]]),
                    y=np.concatenate([np.zeros_like(temperatures), hc1_values[::-1]]),
                    fill='toself',
                    fillcolor='rgba(0, 0, 255, 0.2)',
                    line=dict(color='rgba(0, 0, 255, 0)'),
                    name='Meissner State'
                ))
                phase_fig.add_trace(go.Scatter(
                    x=np.concatenate([temperatures, temperatures[::-1]]),
                    y=np.concatenate([hc1_values, hc2_values[::-1]]),
                    fill='toself',
                    fillcolor='rgba(128, 0, 128, 0.2)',
                    line=dict(color='rgba(128, 0, 128, 0)'),
                    name='Mixed State (Vortex)'
                ))
                if temperature <= tc:
                    if magnetic_field <= current_material["hc1"] * (1 - (temperature/tc)**2):
                        state = "Meissner State"
                        state_color = "blue"
                    elif magnetic_field <= current_material["hc2"] * (1 - (temperature/tc)**2):
                        state = "Mixed State"
                        state_color = "purple"
                    else:
                        state = "Normal"
                        state_color = "red"
                else:
                    state = "Normal"
                    state_color = "red"
                phase_fig.add_trace(go.Scatter(
                    x=[temperature],
                    y=[magnetic_field],
                    mode='markers',
                    marker=dict(size=15, color=state_color, symbol='star'),
                    name='Current State'
                ))

            phase_fig.update_layout(
                title=f'Phase Diagram for {material}',
                xaxis_title='Temperature (K)',
                yaxis_title='Magnetic Field (T)',
                height=500,
                legend=dict(yanchor="top", y=0.99, xanchor="left", x=0.01)
            )
            st.plotly_chart(phase_fig, use_container_width=True)
            st.markdown(f"""
            **Current State: {state}**

            At T = {temperature:.1f} K and H = {magnetic_field:.3f} T, {material} is in the **{state}** phase.
            """)
            with st.expander("About Phase Diagrams"):
                st.markdown("""
                ### Phase Diagrams in Superconductivity

                The phase diagram shows the boundaries between different states of a superconductor:

                **For Type-I Superconductors:**
                - **Superconducting State**: Below both critical temperature (Tc) and critical field (Hc)
                - **Normal State**: Above either Tc or Hc

                **For Type-II Superconductors:**
                - **Meissner State**: Below lower critical field (Hc1) and Tc - complete expulsion of magnetic field
                - **Mixed/Vortex State**: Between Hc1 and upper critical field (Hc2) - magnetic flux penetrates in quantized vortices
                - **Normal State**: Above either Tc or Hc2

                The critical field boundaries follow approximately the relation:

                $$H_c(T) = H_c(0)[1-(T/T_c)^2]$$
                """)

        with meissner_tab:
            st.subheader("Meissner Effect Visualization")
            n_points = 30
            x = np.linspace(-3, 3, n_points)
            y = np.linspace(-3, 3, n_points)
            X, Y = np.meshgrid(x, y)
            radius = 1.5
            inside_sc = X**2 + Y**2 <= radius**2
            U = np.zeros_like(X)
            V = np.zeros_like(Y)

            def create_field_lines(temp, field, material_props):
                material_tc = material_props["tc"]
                is_superconducting = temp < material_tc
                field_strength = min(1.0, field / (0.1 if field < 0.1 else field))
                U = np.zeros_like(X)
                V = np.zeros_like(Y)
                U[~inside_sc] = 0
                V[~inside_sc] = field_strength
                if is_superconducting:
                    if material_props["type"] == "Type-I":
                        if field < material_props["hc"] * (1 - (temp/material_tc)**2):
                            U[inside_sc] = 0
                            V[inside_sc] = 0
                        else:
                            U[inside_sc] = 0
                            V[inside_sc] = field_strength
                    else:
                        hc1_current = material_props["hc1"] * (1 - (temp/material_tc)**2)
                        hc2_current = material_props["hc2"] * (1 - (temp/material_tc)**2)
                        if field < hc1_current:
                            U[inside_sc] = 0
                            V[inside_sc] = 0
                        elif field < hc2_current:
                            penetration_ratio = (field - hc1_current) / (hc2_current - hc1_current)
                            U[inside_sc] = 0
                            V[inside_sc] = field_strength * penetration_ratio
                        else:
                            U[inside_sc] = 0
                            V[inside_sc] = field_strength
                else:
                    U[inside_sc] = 0
                    V[inside_sc] = field_strength
                return U, V

            U, V = create_field_lines(temperature, magnetic_field, current_material)
            meissner_fig = go.Figure()
            meissner_fig.add_trace(go.Streamtube(
                x=X.flatten(),
                y=Y.flatten(),
                z=np.zeros_like(X.flatten()),
                u=U.flatten(),
                v=V.flatten(),
                w=np.zeros_like(X.flatten()),
                sizeref=0.5,
                colorscale='Blues',
                showscale=False
            ))
            theta = np.linspace(0, 2*np.pi, 100)
            x_circle = radius * np.cos(theta)
            y_circle = radius * np.sin(theta)
            z_circle = np.zeros_like(theta)
            meissner_fig.add_trace(go.Scatter3d(
                x=x_circle,
                y=y_circle,
                z=z_circle,
                mode='lines',
                line=dict(color='black', width=4),
                name='Superconductor'
            ))
            meissner_fig.update_layout(
                title=f'Meissner Effect for {material} at {temperature:.1f} K',
                scene=dict(
                    xaxis=dict(range=[-3, 3], showticklabels=False),
                    yaxis=dict(range=[-3, 3], showticklabels=False),
                    zaxis=dict(range=[-1, 1], showticklabels=False),
                    aspectmode='manual',
                    aspectratio=dict(x=1, y=1, z=0.4)
                ),
                height=600,
                width=700
            )
            st.plotly_chart(meissner_fig, use_container_width=True)
            st.markdown(f"""
            **Meissner Effect Visualization**

            This visualization shows magnetic field lines around a superconducting sample. 
            The blue streamlines represent magnetic field lines.

            **Current state:** {state}

            - Temperature: {temperature:.1f} K (Critical temperature: {tc} K)
            - Applied field: {magnetic_field:.3f} T
            """)
            if state == "Meissner State" or (state == "Superconducting" and material_type == "Type-I"):
                st.markdown("""
                The magnetic field is completely expelled from the superconductor, 
                demonstrating the perfect diamagnetism of the Meissner effect.
                """)
            elif state == "Mixed State":
                st.markdown("""
                The magnetic field partially penetrates the superconductor in the form of quantized vortices, 
                each carrying a magnetic flux quantum Φ₀ = h/2e ≈ 2.07 × 10⁻¹⁵ Wb.
                """)
            else:
                st.markdown("""
                The material is in the normal state, allowing magnetic field to penetrate through it.
                """)
            with st.expander("About the Meissner Effect"):
                st.markdown("""
                ### The Meissner Effect

                The Meissner effect is the complete expulsion of magnetic fields from a superconductor during its transition to the superconducting state.

                Key aspects:

                - It's distinct from perfect conductivity: a perfect conductor would trap magnetic flux, while a superconductor expels it
                - The expulsion creates screening currents on the surface of the superconductor
                - The magnetic field decays exponentially inside the superconductor with the characteristic London penetration depth (λ)
                - Type-I superconductors show complete expulsion below Hc
                - Type-II superconductors show complete expulsion below Hc1, and partial penetration via vortices between Hc1 and Hc2

                The magnetic penetration depth is given by:

                $$λ = \\sqrt{m / (μ_0 n_s e^2)}$$

                where m is the electron mass, n_s is the density of superconducting electrons, e is the electron charge, and μ₀ is the vacuum permeability.
                """)

        with cooper_tab:
            st.subheader("Cooper Pair Formation")
            kb = 8.617e-5  # Boltzmann constant in eV/K
            delta_0 = 1.76 * kb * tc  # in eV
            if temperature < tc:
                delta_T = delta_0 * np.sqrt(1 - (temperature/tc))
            else:
                delta_T = 0
            coherence_length = 50 if material_type == "Type-II" else 100  # nm

            st.markdown(f"""
            **Cooper Pair Visualization**

            Temperature: {temperature:.1f} K

            Superconducting gap: {delta_T*1000:.2f} meV

            Coherence length: ~{coherence_length} nm
            """)

            def create_cooper_pair_animation(temp, tc, frames=30):
                t = np.linspace(0, 2*np.pi, frames)
                sc_ratio = max(0, 1 - temp/tc) if tc > 0 else 0
                fig = go.Figure(frames=[
                    go.Frame(
                        data=[
                            go.Scatter3d(
                                x=np.array([0, 1, 2, 0, 1, 2, 0, 1, 2]),
                                y=np.array([0, 0, 0, 1, 1, 1, 2, 2, 2]),
                                z=np.zeros(9),
                                mode='markers',
                                marker=dict(size=10, color='red'),
                                name='Lattice Ions'
                            ),
                            go.Scatter3d(
                                x=[1 + 0.2*np.sin(time + np.pi)],
                                y=[1 + 0.2*np.cos(time + np.pi)],
                                z=[0.1 + 0.05*np.sin(2*time)],
                                mode='markers',
                                marker=dict(size=6, color='blue'),
                                name='Electron 1'
                            ),
                            go.Scatter3d(
                                x=[1 + 0.2*np.sin(time)],
                                y=[1 + 0.2*np.cos(time)],
                                z=[0.1 + 0.05*np.sin(2*time + np.pi)],
                                mode='markers',
                                marker=dict(size=6, color='blue'),
                                name='Electron 2'
                            ),
                            go.Scatter3d(
                                x=[1 + r*np.cos(a) for r in np.linspace(0, 0.3*sc_ratio, 10) for a in [time, time+np.pi]],
                                y=[1 + r*np.sin(a) for r in np.linspace(0, 0.3*sc_ratio, 10) for a in [time, time+np.pi]],
                                z=[0.1 for _ in range(20)],
                                mode='lines',
                                line=dict(color='green', width=2, dash='dot') if sc_ratio > 0 else dict(color='rgba(0,0,0,0)'),
                                name='Phonon Interaction'
                            )
                        ],
                        name=f"frame{i}"
                    )
                    for i, time in enumerate(t)
                ])

                fig.add_trace(go.Scatter3d(
                    x=np.array([0, 1, 2, 0, 1, 2, 0, 1, 2]),
                    y=np.array([0, 0, 0, 1, 1, 1, 2, 2, 2]),
                    z=np.zeros(9),
                    mode='markers',
                    marker=dict(size=10, color='red'),
                    name='Lattice Ions'
                ))
                fig.add_trace(go.Scatter3d(
                    x=[1 + 0.2*np.sin(0 + np.pi)],
                    y=[1 + 0.2*np.cos(0 + np.pi)],
                    z=[0.1],
                    mode='markers',
                    marker=dict(size=6, color='blue'),
                    name='Electron 1'
                ))
                fig.add_trace(go.Scatter3d(
                    x=[1 + 0.2*np.sin(0)],
                    y=[1 + 0.2*np.cos(0)],
                    z=[0.1],
                    mode='markers',
                    marker=dict(size=6, color='blue'),
                    name='Electron 2'
                ))
                if sc_ratio > 0:
                    fig.add_trace(go.Scatter3d(
                        x=[1, 1],
                        y=[1, 1],
                        z=[0.1, 0.1],
                        mode='lines',
                        line=dict(color='green', width=2, dash='dot'),
                        name='Phonon Interaction'
                    ))
                fig.update_layout(
                    updatemenus=[{
                        'type': 'buttons',
                        'showactive': False,
                        'buttons': [{
                            'label': 'Play',
                            'method': 'animate',
                            'args': [None, {'frame': {'duration': 100, 'redraw': True}, 'fromcurrent': True}]
                        }]
                    }],
                    title='Cooper Pair Formation',
                    scene=dict(
                        xaxis=dict(range=[-0.5, 2.5], showticklabels=False),
                        yaxis=dict(range=[-0.5, 2.5], showticklabels=False),
                        zaxis=dict(range=[-0.2, 0.5], showticklabels=False),
                        aspectmode='manual',
                        aspectratio=dict(x=1, y=1, z=0.5)
                    ),
                    height=PLOT_HEIGHT,
                    width=PLOT_WIDTH
                )
                return fig

            cooper_fig = create_cooper_pair_animation(temperature, tc)
            st.plotly_chart(cooper_fig, use_container_width=True)

            # Fixed LaTeX formatting using raw string
            st.markdown(r"""
            This animation conceptually illustrates Cooper pair formation via phonon exchange:

            1. Electron 1 (blue) distorts the lattice of positive ions (red)
            2. This distortion propagates as a phonon (green dotted lines)
            3. Electron 2 (blue) is attracted to the resulting positive charge density
            4. The two electrons form a Cooper pair, despite their like charges

            In BCS theory, the critical temperature is related to the electron-phonon coupling strength:

            $$T_c \approx 1.14 \Theta_D e^{-1/(N_0 V)}$$

            Where:
            - $\Theta_D$: Debye temperature
            - $N_0$: Density of states at the Fermi level
            - $V$: Electron-phonon coupling constant
            """)

            st.subheader("Superconducting Energy Gap")
            energies = np.linspace(-5*delta_0, 5*delta_0, 1000)
            dos_normal = np.ones_like(energies)
            dos_sc = np.zeros_like(energies)
            if delta_T > 0:
                valid_indices = np.abs(energies) >= delta_T
                dos_sc[valid_indices] = np.abs(energies[valid_indices]) / np.sqrt(energies[valid_indices]**2 - delta_T**2)

            dos_fig = go.Figure()
            dos_fig.add_trace(go.Scatter(
                x=energies/delta_0,
                y=dos_normal,
                mode='lines',
                name='Normal State',
                line=dict(color=COLOR_SCHEME['normal'], width=2, dash='dash')
            ))
            dos_fig.add_trace(go.Scatter(
                x=energies/delta_0,
                y=dos_sc,
                mode='lines',
                name='Superconducting State',
                line=dict(color=COLOR_SCHEME['superconducting'], width=3)
            ))
            dos_fig.update_layout(
                title=f'Density of States at T = {temperature:.1f} K',
                xaxis_title='Energy (E/Δ₀)',
                yaxis_title='Density of States (N(E)/N₀)',
                height=PLOT_HEIGHT,
                yaxis=dict(range=[0, 5]),
                legend=dict(yanchor="top", y=0.99, xanchor="right", x=0.99)
            )
            st.plotly_chart(dos_fig, use_container_width=True)

            # Fixed LaTeX formatting using raw string with f-string variables
            st.markdown(fr"""
            This plot shows the density of states (DOS) for electrons in:
            - Normal state (red dashed line): constant DOS near the Fermi level
            - Superconducting state (blue line): energy gap of 2Δ = {2*delta_T*1000:.2f} meV around the Fermi level

            The superconducting energy gap is temperature-dependent, following:

            $$\Delta(T) \approx \Delta(0)\sqrt{{1 - \frac{{T}}{{T_c}}}}$$

            At T = {temperature:.1f} K, the gap is {delta_T/delta_0:.2%} of its zero-temperature value.
            """)

            with st.expander("About BCS Theory"):
                st.markdown(r"""
                ### BCS Theory Fundamentals

                The Bardeen-Cooper-Schrieffer (BCS) theory explains the microscopic mechanism of superconductivity:

                1. **Cooper Pair Formation**: Electrons with opposite momenta and spins pair up due to phonon-mediated attraction
                2. **Energy Gap**: A gap of 2$\Delta$ opens around the Fermi energy in the electronic spectrum
                3. **Key Relationships**:
                    - $\Delta(0) = 1.76 k_B T_c$
                    - Coherence length: $\xi \approx \hbar v_F / \pi \Delta$
                    - Critical temperature: $T_c \approx 1.14 \Theta_D e^{-1/(N(0)V)}$
                4. **Type-I vs Type-II**:
                    - Type-I: $\xi > \lambda$ (coherence length > penetration depth)
                    - Type-II: $\xi < \lambda$ (coherence length < penetration depth)
                5. **Flux Quantization**: Magnetic flux through a superconducting loop is quantized in units of:
                    - $\Phi_0 = h/2e \approx 2.07 \times 10^{-15}$ weber
                """)

    st.header("Electrical Resistivity")
    temps = np.linspace(0, tc*1.5, 100)
    residual_resistivity = 0.02
    resistivity = np.zeros_like(temps)

    for i, t in enumerate(temps):
        if t < tc:
            resistivity[i] = residual_resistivity * np.exp(-(tc-t)/tc * 5)
        else:
            resistivity[i] = 0.1 + 0.02 * (t-tc)

    resistivity_fig = go.Figure()
    resistivity_fig.add_trace(go.Scatter(
        x=temps,
        y=resistivity,
        mode='lines',
        line=dict(color=COLOR_SCHEME['superconducting'], width=3)
    ))
    resistivity_fig.add_vline(
        x=tc, 
        line_dash="dash", 
        line_color=COLOR_SCHEME['normal'],
        annotation_text=f"Tc = {tc} K",
        annotation_position="top right"
    )
    resistivity_fig.add_vline(
        x=temperature,
        line_dash="dot", 
        line_color='green'
    )
    resistivity_fig.update_layout(
        title=f'Resistivity vs Temperature for {material}',
        xaxis_title='Temperature (K)',
        yaxis_title='Resistivity (relative units)',
        height=PLOT_HEIGHT
    )
    st.plotly_chart(resistivity_fig, use_container_width=True)

    st.markdown(f"""
    **Zero Resistance Property**

    This plot demonstrates the hallmark property of superconductors: the sudden drop to zero electrical resistance below Tc.

    Current temperature ({temperature:.1f} K) is marked with a green line.

    The resistance drop occurs because:
    1. Cooper pairs form a coherent quantum state
    2. This state can move through the crystal without scattering
    3. The energy gap prevents electron excitations that would cause resistance
    """)

    st.header("Applications of Superconductivity")
    col1, col2, col3 = st.columns(3)

    with col1:
        st.image("https://upload.wikimedia.org/wikipedia/commons/thumb/2/2f/Superconducting_magnet_bank.jpg/320px-Superconducting_magnet_bank.jpg", caption="Superconducting Magnets")
        st.markdown("""
        **Superconducting Magnets**
        - MRI machines
        - Particle accelerators
        - Nuclear fusion reactors
        """)

    with col2:
        st.image("https://upload.wikimedia.org/wikipedia/commons/thumb/9/9a/SQUID_Magnetometer_at_University_of_Rome%2C_Sapienza.jpg/320px-SQUID_Magnetometer_at_University_of_Rome%2C_Sapienza.jpg", caption="SQUIDs")
        st.markdown("""
        **SQUIDs**
        - Ultra-sensitive magnetometers
        - Brain activity measurements
        - Quantum computing
        """)

    with col3:
        st.image("https://upload.wikimedia.org/wikipedia/commons/thumb/0/04/Maglev_June_2005.jpg/320px-Maglev_June_2005.jpg", caption="Maglev Trains")
